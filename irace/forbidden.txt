## Template for specifying forbidden parameter configurations in irace.
##
## This filename must be specified via the --forbidden-file command-line option
## (or forbiddenFile in scenario.txt).
##
## The format is one constraint per line. Each constraint is a logical
## expression (in R syntax). If a parameter configuration
## is generated that makes the logical expression evaluate to TRUE,
## then the configuration is discarded.
##
## Examples of valid logical operators are: ==  !=  >=  <=  >  <  &  |  !  %in%
## param1 < 5 & mode == "x1"
## (param2 > 6 & mode == "x1") | (param2 <= 6 & mode == "x2")
## real < 4 & mode %in% c("x2", "x3")

# Direct front mapping requires the ParetoFronts selector.
pheromoneType %in% c("TwoDim", "ItemTwoDim") & updateType == "PartFromParetoFront" & solutionsSelectionStrategy != "ParetoFronts"

# PairingCombination requires an even bucket count; direct front mapping derives it from the fronts.
# The selector and table guards allow inactive parameters for Basic tables and selections other than ParetoFronts.
pheromoneType %in% c("TwoDim", "ItemTwoDim") & getType == "PairingCombination" & solutionsSelectionStrategy == "ParetoFronts" & updateType == "PartFromParetoFront" & paretoFrontsCount %% 2 == 1

# For Index/Evaluation, parity applies to the independent size, not to the archive's front count.
pheromoneType %in% c("TwoDim", "ItemTwoDim") & getType == "PairingCombination" & updateType %in% c("PartFromIndex", "PartFromEvaluation") & twoDimPhSize %% 2 == 1
